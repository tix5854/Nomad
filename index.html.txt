        // Filter Products Array Based on Active State
        function getFilteredProducts() {
            return products.filter(p => {
                // Must be approved unless viewing admin
                if(p.status !== 'Approved' && state.currentRole !== 'admin') return false;

                // Category Filter
                if(state.activeCategory !== 'all' && p.category !== state.activeCategory) return false;

                // Location Filter
                if(state.activeLocation !== 'all' && p.location !== state.activeLocation) return false;

                // Price Filter
                if(p.price > state.priceLimit) return false;

                // Search Query Filter
                if(state.searchQuery) {
                    const q = state.searchQuery.toLowerCase();
                    const matchTitle = p.title.toLowerCase().includes(q);
                    const matchSeller = p.sellerName.toLowerCase().includes(q);
                    const matchLoc = p.location.toLowerCase().includes(q);
                    if(!matchTitle && !matchSeller && !matchLoc) return false;
                }

                // Verified SME check
                if(state.verifiedOnly && !p.isVerifiedSME) return false;

                // Made in Namibia check
                if(state.madeInNaOnly && !p.isMadeInNA) return false;

                return true;
            }).sort((a, b) => {
                if(state.sortOption === 'price-low') return a.price - b.price;
                if(state.sortOption === 'price-high') return b.price - a.price;
                if(state.sortOption === 'rating') return b.rating - a.rating;
                if(state.sortOption === 'newest') return b.id - a.id;
                return 0; // Default featured
            });
        }

        function switchView(viewName) {
            state.currentView = viewName;
            const hero = document.getElementById('hero-section');
            const dynamicContainer = document.getElementById('dynamic-view-container');
            const mainGridSection = document.querySelector('section.py-10');
            const topSellers = document.querySelector('section.py-12');

            // Reset Subtitles
            document.getElementById('section-title').innerText = "Featured Local Products";

            if(viewName === 'home') {
                hero.classList.remove('hidden');
                mainGridSection.classList.remove('hidden');
                topSellers.classList.remove('hidden');
                dynamicContainer.classList.add('hidden');
                resetFilters();
            } else {
                hero.classList.add('hidden');
                mainGridSection.classList.add('hidden');
                topSellers.classList.add('hidden');
                dynamicContainer.classList.remove('hidden');

                if(viewName === 'seller') renderSellerDashboard();
                else if(viewName === 'admin') renderAdminPanel();
                else if(viewName === 'shops') renderSmeDirectoryView();
                else if(viewName === 'wishlist') renderWishlistView();
                else if(viewName === 'services') renderServicesView();
            }
            window.scrollTo({ top: 0, behavior: 'smooth' });
        }

        function addToCart(productId) {
            const product = products.find(p => p.id === productId);
            if(!product) return;

            const existing = state.cart.find(item => item.id === productId);
            if(existing) {
                existing.qty++;
            } else {
                state.cart.push({ ...product, qty: 1 });
            }
            updateCartBadge();
            renderCartDrawerItems();
            
            // Visual Feedback
            const cartBtn = document.querySelector('button[onclick="toggleCartDrawer()"]');
            cartBtn.classList.add('scale-110');
            setTimeout(() => cartBtn.classList.remove('scale-110'), 200);
        }

        function updateCartBadge() {
            const count = state.cart.reduce((acc, item) => acc + item.qty, 0);
            const total = state.cart.reduce((acc, item) => acc + (item.price * item.qty), 0);
            
            document.getElementById('cart-count').innerText = count;
            document.getElementById('cart-total-header').innerText = `N$ ${total}`;
            document.getElementById('cart-subtotal-val').innerText = `N$ ${total.toFixed(2)}`;
        }

        function toggleCartDrawer() {
            const drawer = document.getElementById('cart-drawer');
            drawer.classList.toggle('hidden');
            if(!drawer.classList.contains('hidden')) {
                renderCartDrawerItems();
            }
        }

        function renderCartDrawerItems() {
            const container = document.getElementById('cart-items-container');
            if(state.cart.length === 0) {
                container.innerHTML = `
                    <div class="text-center py-10 text-gray-400">
                        <i class="fa-solid fa-basket-shopping text-3xl mb-2"></i>
                        <p class="text-xs">Your basket is currently empty.</p>
                    </div>
                `;
                return;
            }

            container.innerHTML = state.cart.map(item => `
                <div class="flex items-center justify-between p-3 rounded-2xl bg-gray-50 dark:bg-slate-700/50 border border-gray-200 dark:border-slate-600">
                    <img src="${item.images[0]}" alt="${item.title}" class="w-12 h-12 rounded-xl object-cover">
                    <div class="flex-1 px-3 min-w-0">
                        <h4 class="text-xs font-bold text-gray-800 dark:text-white truncate">${item.title}</h4>
                        <p class="text-xs text-nomad-blue dark:text-blue-400 font-bold">N$ ${item.price}</p>
                    </div>
                    <div class="flex items-center space-x-2">
                        <button onclick="changeCartQty(${item.id}, -1)" class="w-6 h-6 rounded-lg bg-gray-200 dark:bg-slate-600 text-xs font-bold dark:text-white">-</button>
                        <span class="text-xs font-bold dark:text-white">${item.qty}</span>
                        <button onclick="changeCartQty(${item.id}, 1)" class="w-6 h-6 rounded-lg bg-gray-200 dark:bg-slate-600 text-xs font-bold dark:text-white">+</button>
                    </div>
                </div>
            `).join('');
        }

        function changeCartQty(id, delta) {
            const item = state.cart.find(i => i.id === id);
            if(!item) return;
            item.qty += delta;
            if(item.qty <= 0) {
                state.cart = state.cart.filter(i => i.id !== id);
            }
            updateCartBadge();
            renderCartDrawerItems();
        }

        function openCheckoutModal() {
            if(state.cart.length === 0) return;
            toggleCartDrawer();
            const total = state.cart.reduce((acc, item) => acc + (item.price * item.qty), 0);
            document.getElementById('chk-items-total').innerText = `N$ ${total.toFixed(2)}`;
            document.getElementById('chk-grand-total').innerText = `N$ ${(total + 45).toFixed(2)}`;
            
            document.getElementById('checkout-step-1').classList.remove('hidden');
            document.getElementById('checkout-step-2').classList.add('hidden');
            document.getElementById('checkout-modal').classList.remove('hidden');
        }

            document.getElementById('checkout-modal').classList.add('hidden');
        }

        function toggleWishlist(id, e) {
            if(e) e.stopPropagation();
            if(state.wishlist.includes(id)) {
                state.wishlist = state.wishlist.filter(item => item !== id);
            } else {
                state.wishlist.push(id);
            }
            updateWishlistBadge();
            renderProductGrid();
            if(state.currentView === 'wishlist') renderWishlistView();
        }

        function updateWishlistBadge() {
            const badge = document.getElementById('wishlist-badge');
            if(state.wishlist.length > 0) {
                badge.innerText = state.wishlist.length;
                badge.classList.remove('hidden');
            } else {
                badge.classList.add('hidden');
            }
        }

        /* ==========================================================================
           9. FILTER & SEARCH HANDLERS
           ========================================================================== */

        function filterCategory(catId) {
            state.activeCategory = catId;
            renderCategories();
            switchView('home');
            applyFilters();
        }

        function filterSpecial(type) {
            switchView('home');
            if(type === 'made-in-namibia') {
                state.madeInNaOnly = true;
                document.getElementById('section-title').innerText = "Made in Namibia Showcase";
            } else if(type === 'farmers-market') {
                state.activeCategory = 'Agriculture';
                renderCategories();
                document.getElementById('section-title').innerText = "Local Farmers Harvest Market";
            }
            applyFilters();
        }

        function filterSellerProducts(sellerName) {
            switchView('home');
            state.searchQuery = sellerName;
            document.getElementById('section-title').innerText = "Products by " + sellerName;
            applyFilters();
        }

        function updatePriceLabel(val) {
            state.priceLimit = parseFloat(val);
            document.getElementById('price-limit-val').innerText = `N$ ${val}`;
        }

        function applyFilters() {
            state.activeLocation = document.getElementById('filter-location').value;
            state.sortOption = document.getElementById('sort-select').value;
            state.verifiedOnly = document.getElementById('filter-verified-only').checked;
            state.madeInNaOnly = document.getElementById('filter-made-in-na').checked;
            renderProductGrid();
        }

        function resetFilters() {
            state.activeCategory = 'all';
            state.activeLocation = 'all';
            state.searchQuery = '';
            state.priceLimit = 5000;
            state.verifiedOnly = false;
            state.madeInNaOnly = false;
            
            document.getElementById('filter-location').value = 'all';
            document.getElementById('filter-price').value = 5000;
            document.getElementById('price-limit-val').innerText = 'N$ 5,000';
            document.getElementById('filter-verified-only').checked = false;
            document.getElementById('filter-made-in-na').checked = false;
            document.getElementById('global-search').value = '';

            renderCategories();
            renderProductGrid();
        }

        function handleGlobalSearch(e) {
            if(e.key === 'Enter') executeSearch();
        }

        function executeSearch() {
            const query = document.getElementById('global-search').value;
            state.searchQuery = query;
            switchView('home');
            applyFilters();
        }

        function executeHeroSearch() {
            const query = document.getElementById('hero-search-input').value;
            const loc = document.getElementById('hero-location-select').value;
            state.searchQuery = query;
            state.activeLocation = loc;
            document.getElementById('filter-location').value = loc;
            applyFilters();
        }

           10. REPORTING SYSTEM
           ========================================================================== */

        function openReportModal(productId) {
            document.getElementById('report-prod-id').value = productId;
            document.getElementById('report-modal').classList.remove('hidden');
        }

        function closeReportModal() {
            document.getElementById('report-modal').classList.add('hidden');
        }

        function submitReport(e) {
            e.preventDefault();
            const pId = parseInt(document.getElementById('report-prod-id').value);
            const product = products.find(p => p.id === pId);

            const newReport = {
                id: Date.now(),
                productId: pId,
                productName: product ? product.title : 'Item #' + pId,
                reason: document.getElementById('report-reason').value,
                details: document.getElementById('report-details').value,
                reporter: 'user_na@gmail.com',
                status: 'Pending Review'
            };

            state.reports.push(newReport);
            closeReportModal();
            closeProductModal();
            alert('Thank you. Your report has been submitted to Nomad moderators for review.');
        }
